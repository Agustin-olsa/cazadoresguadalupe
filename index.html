          jabalies,

          otros,

          total,

          creadoEn: serverTimestamp()

        });

        msgEl.className = "msg-ok";

        msgEl.textContent = `Montería guardada (${total} piezas).`;

        document.getElementById("monteria-nombre").value = "";

        document.getElementById("monteria-venados").value = 0;

        document.getElementById("monteria-cervuno").value = 0;

        document.getElementById("monteria-jabalies").value = 0;

        document.getElementById("monteria-otros").value = "";

      } catch (err) {

        msgEl.className = "msg-error";

        msgEl.textContent = "Error al guardar en Firebase.";

      }

    });


    // Guardar Espera / Rececho en Firestore

    formER.addEventListener("submit", async (e) => {

      e.preventDefault();

      const msgEl = document.getElementById("er-msg");

      msgEl.className = "";

      msgEl.textContent = "Guardando...";


      const fecha = document.getElementById("er-fecha").value;

      const accion = erAccion.value;

      const anio = fecha ? Number(fecha.split("-")[0]) : new Date().getFullYear();


      let especie = "";

      let cantidad = 0;


      if (accion === "Espera") {

        especie = "Jabalí";

        cantidad = Number(document.getElementById("er-jabali-num").value) || 0;

      } else {

        especie = document.getElementById("er-especie-rececho").value; // Venado o Corzo

        cantidad = Number(document.getElementById("er-rececho-num").value) || 0;

      }


      try {

        await addDoc(collection(db, "capturas"), {

          tipo: "er",

          fecha,

          anio,

          accion,

          especie,

          cantidad,

          total: cantidad,

          creadoEn: serverTimestamp()

        });

        msgEl.className = "msg-ok";

        msgEl.textContent = `${accion} guardada (${cantidad} ${especie}).`;

        document.getElementById("er-jabali-num").value = 1;

        document.getElementById("er-rececho-num").value = 1;

      } catch (err) {

        msgEl.className = "msg-error";

        msgEl.textContent = "Error al guardar en Firebase.";

      }

    });


    // Escuchar datos en tiempo real desde Firestore

    function subscribeToCapturas() {

      const q = query(collection(db, "capturas"), orderBy("fecha", "desc"));

      unsubscribeRecords = onSnapshot(q, (snapshot) => {

        allRecords = snapshot.docs.map((d) => ({ id: d.id, ...d.data() }));

        populateYearFilter();

        renderStatsAndList();

      });

    }


    function populateYearFilter() {

      const currentSelected = filterYear.value;

      const years = new Set(["2026", "2027"]);

      allRecords.forEach((r) => {

        if (r.anio) years.add(String(r.anio));

      });

      const sortedYears = Array.from(years).sort((a, b) => Number(b) - Number(a));


      filterYear.innerHTML = `<option value="todos">Todos los años</option>`;

      sortedYears.forEach((y) => {

        const opt = document.createElement("option");

        opt.value = y;

        opt.textContent = `Temporada ${y}`;

        filterYear.appendChild(opt);

      });


      if (Array.from(filterYear.options).some((o) => o.value === currentSelected)) {

        filterYear.value = currentSelected;

      }

    }


    // Filtros de listado

    document.querySelectorAll(".list-filter-btn").forEach((btn) => {

      btn.addEventListener("click", () => {

        document.querySelectorAll(".list-filter-btn").forEach((b) => b.classList.remove("active"));

        btn.classList.add("active");

        currentListFilter = btn.dataset.listFilter;

        renderStatsAndList();

      });

    });


    filterYear.addEventListener("change", renderStatsAndList);

    sortBy.addEventListener("change", renderStatsAndList);


    function renderStatsAndList() {

      const selectedYear = filterYear.value;

      const yearFiltered = selectedYear === "todos"

        ? allRecords

        : allRecords.filter((r) => String(r.anio) === selectedYear);


      const totalMonterias = yearFiltered

        .filter((r) => r.tipo === "monteria")

        .reduce((acc, r) => acc + (Number(r.total) || 0), 0);


      const totalER = yearFiltered

        .filter((r) => r.tipo === "er")

        .reduce((acc, r) => acc + (Number(r.total) || 0), 0);


      const totalGeneral = totalMonterias + totalER;


      document.getElementById("stat-total-general").textContent = totalGeneral;

      document.getElementById("stat-total-monterias").textContent = totalMonterias;

      document.getElementById("stat-total-er").textContent = totalER;


      // Desglose por año/temporada

      const byYear = {};

      allRecords.forEach((r) => {

        const y = r.anio || "Sin año";

        if (!byYear[y]) byYear[y] = { total: 0, monterias: 0, er: 0 };

        const val = Number(r.total) || 0;

        byYear[y].total += val;

        if (r.tipo === "monteria") byYear[y].monterias += val;

        if (r.tipo === "er") byYear[y].er += val;

      });


      const breakdownEl = document.getElementById("season-breakdown");

      const yearKeys = Object.keys(byYear).sort((a, b) => Number(b) - Number(a));

      breakdownEl.innerHTML = yearKeys.length

        ? yearKeys.map((y) =>

            `<div>• <strong>Año ${y}:</strong> ${byYear[y].total} piezas (Monterías: ${byYear[y].monterias} | Esperas/Recechos: ${byYear[y].er})</div>`

          ).join("")

        : "<div>Aún no hay registros guardados.</div>";


      // Filtrar por tipo para la lista y ordenar

      let listItems = currentListFilter === "todas"

        ? [...yearFiltered]

        : yearFiltered.filter((r) => r.tipo === currentListFilter);


      listItems.sort((a, b) => {

        const totA = Number(a.total) || 0;

        const totB = Number(b.total) || 0;

        if (sortBy.value === "total_desc") return totB - totA;

        if (sortBy.value === "total_asc") return totA - totB;

        if (sortBy.value === "fecha_desc") return (b.fecha || "").localeCompare(a.fecha || "");

        if (sortBy.value === "fecha_asc") return (a.fecha || "").localeCompare(b.fecha || "");

        return 0;

      });


      if (!listItems.length) {

        recordsList.innerHTML = `<p class="record-meta">No hay registros para este filtro.</p>`;

        return;

      }


      recordsList.innerHTML = listItems.map((item) => {

        if (item.tipo === "monteria") {

          return `

            <div class="record-item">

              <div class="record-top">

                <span class="badge">MONTERÍA · ${item.total} piezas</span>

                <button class="btn btn-danger btn-small" onclick="window.eliminarCaptura('${item.id}')">Borrar</button>

              </div>

              <div class="record-meta">

                <strong>${item.nombre || "Montería"}</strong> — Fecha: <strong>${item.fecha}</strong> (Año ${item.anio})<br />

                Venados: <strong>${item.venados}</strong> · Cervuno: <strong>${item.cervuno}</strong> · Jabalís: <strong>${item.jabalies}</strong>

                ${item.otros ? `<br />Otros: ${item.otros}` : ""}

              </div>

            </div>

          `;

        } else {

          return `

            <div class="record-item">

              <div class="record-top">

                <span class="badge">${item.accion.toUpperCase()} · ${item.total} ${item.especie}</span>

                <button class="btn btn-danger btn-small" onclick="window.eliminarCaptura('${item.id}')">Borrar</button>

              </div>

              <div class="record-meta">

                Fecha: <strong>${item.fecha}</strong> (Año ${item.anio})<br />

                Acción: <strong>${item.accion}</strong> — ${item.especie}: <strong>${item.cantidad}</strong>

              </div>

            </div>

          `;

        }

      }).join("");

    }


    window.eliminarCaptura = async (id) => {

      if (confirm("¿Seguro que quieres eliminar este registro?")) {

        await deleteDoc(doc(db, "capturas", id));

      }

    };
